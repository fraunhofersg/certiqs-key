\begin{theorem}\label{thm:cond}
For a base object of Definition~\ref{def:base} and a fresh object $z$ of
Definition~\ref{def:fresh}, $x \le 1 + z$.
\end{theorem}

\begin{proof}
By \eqref{eq:base} and $z \ge 0$.
\end{proof}
